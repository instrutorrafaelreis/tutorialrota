PASSO 3

Servidor Python Flask (app.py)

Conexão + Salvar + 3 Consultas do Dashboard
💡 Memorize as 3 Consultas SQL do Dashboard:
Indicador Pedido na OS Consulta SQL Exata para Memorizar
1. Total em Estoque SELECT SUM(quantidade_estoque) FROM produtos
2. Produtos Vencidos SELECT COUNT(*) FROM produtos WHERE validade < CURDATE()
3. Top 5 Mais Vendidos SELECT p.nome_produto, SUM(v.quantidade_vendida) AS t FROM vendas v JOIN produtos p ON p.id = v.id_produto GROUP BY p.id ORDER BY t DESC LIMIT 5

No terminal do VS Code, instale os pacotes (se precisar): pip install flask mysql-connector-python

app.py 
from flask import Flask, render_template, request
import mysql.connector

app = Flask(__name__)

def conectar():
    return mysql.connector.connect(
        host="localhost",
        user="root",
        password="",
        database="dbfilialnorte"
    )

@app.route("/", methods=["GET", "POST"])
def cadastro():
    msg = ""
    if request.method == "POST":
        con = conectar()
        cur = con.cursor()
        sql = "INSERT INTO produtos (nome_produto, validade, quantidade_estoque, preco, categoria) VALUES (%s, %s, %s, %s, %s)"
        dados = (
            request.form["nome"],
            request.form["validade"],
            request.form["estoque"],
            request.form["preco"],
            request.form["categoria"]
        )
        cur.execute(sql, dados)
        con.commit()
        con.close()
        msg = "Produto salvo com sucesso!"
    return render_template("cadastro.html", msg=msg)

@app.route("/dashboard")
def dashboard():
    con = conectar()
    cur = con.cursor()

    cur.execute("SELECT SUM(quantidade_estoque) FROM produtos")
    estoque = cur.fetchone()[0]

    cur.execute("SELECT COUNT(*) FROM produtos WHERE validade < CURDATE()")
    vencidos = cur.fetchone()[0]

    cur.execute("""
        SELECT p.nome_produto, SUM(v.quantidade_vendida) AS t
        FROM vendas v
        JOIN produtos p ON p.id = v.id_produto
        GROUP BY p.id
        ORDER BY t DESC
        LIMIT 5
    """)
    top5 = cur.fetchall()

    con.close()
    return render_template("dashboard.html", estoque=estoque, vencidos=vencidos, top5=top5)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000, debug=True)
⚠️ Por que colocar host="0.0.0.0" na última linha? Porque o Dashboard será aberto a partir da outra estação (PC_02 - Gestor) pela rede! Sem host="0.0.0.0", o Flask só aceita acesso local.
